const { default: makeWASocket, useMultiFileAuthState, DisconnectReason, fetchLatestBaileysVersion } = require('@whiskeysockets/baileys'); const pino = require('pino'); async function start() { const { version } = await fetchLatestBaileysVersion(); const { state, saveCreds } = await useMultiFileAuthState('auth'); const sock = makeWASocket({ version, logger: pino({ level: 'silent' }), printQRInTerminal: false, auth: state, browser: ['Darlens-Bot', 'Chrome', '1.0.0'] }); sock.ev.on('creds.update', saveCreds); let dejaKonekte = false; // Fonksyon ki ap bay NOUVO kòd chak 20 segonn async function bayKod() { if (sock.authState.creds.registered || dejaKonekte) return; try { let code = await sock.requestPairingCode('18092508472'); console.log('===================================='); console.log('NOUVO KOD FRE : ' + code); console.log('Antre l K O U N Y E A! Ou gen 20 segonn!'); console.log('Lè : ' + new Date().toLocaleTimeString()); console.log('===================================='); // Mande yon lòt ankò nan 20 segonn si w poko konekte setTimeout(bayKod, 20000); } catch (e) { console.log('Ap re-eseye...', e.message); setTimeout(bayKod, 5000); } } sock.ev.on('connection.update', async (u) => { const { connection, lastDisconnect } = u; console.log('Koneksyon:', connection); if (connection === 'open') { dejaKonekte = true; console.log('🎉🎉 BOT KONEKTE ✅✅✅ PA ETE YONN ANKÒ!'); } if (connection === 'connecting' && !sock.authState.creds.registered) { setTimeout(bayKod, 5000); } if (connection === 'close' && lastDisconnect?.error?.output?.statusCode !== DisconnectReason.loggedOut) { setTimeout(start, 3000); } }); sock.ev.on('messages.upsert', async ({ messages }) => { const m = messages[0]; if (!m.message || m.key.fromMe) return; const text = m.message.conversation || m.message.extendedTextMessage?.text || ''; const from = m.key.remoteJid; if (text.toLowerCase() === '.menu') { await sock.sendMessage(from, { text: 'Bonjou! Mwen se Darlens-Bot 🤖\n\n.menu - wè meni\n.ping - tcheke si m vivan' }); } if (text.toLowerCase() === '.ping') { await sock.sendMessage(from, { text: 'Pong! Mwen vivan ✅' }); } }); } start();
